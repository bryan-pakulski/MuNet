cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/pyproject.toml" _munet_version_line REGEX "^version = ")
string(REGEX REPLACE ".*\"([^\"]+)\".*" "\\1" MUNET_VERSION "${_munet_version_line}")
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" MUNET_ABI_VERSION "${MUNET_VERSION}")
project(MuNet VERSION ${MUNET_ABI_VERSION} LANGUAGES CXX)
include(GNUInstallDirs)
option(MUNET_VULKAN "Build the Vulkan compute runtime" ON)
option(MUNET_PYTHON "Build the Python interface" ON)
option(MUNET_SWARM_NODE "Build the standalone Linux/POSIX swarm worker" OFF)
option(MUNET_INSTALL_SDK "Install the C++ library, headers, and CMake package" ON)
add_library(munet_core STATIC cpp/core.cpp cpp/ops.cpp cpp/vulkan.cpp)
file(GLOB MUNET_KERNELS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cpp/kernels/*.inc")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${MUNET_KERNELS})
set(MUNET_KERNEL_SOURCE "")
foreach(kernel IN LISTS MUNET_KERNELS)
  get_filename_component(kernel_name "${kernel}" NAME_WE)
  file(READ "${kernel}" kernel_text)
  string(APPEND MUNET_KERNEL_SOURCE "{\"${kernel_name}\", R\"kernel(${kernel_text})kernel\"},\n")
endforeach()
configure_file(cmake/kernel_sources.hpp.in kernel_sources.hpp @ONLY)
target_include_directories(munet_core PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
add_library(MuNet::core ALIAS munet_core)
target_compile_features(munet_core PUBLIC cxx_std_17)
target_include_directories(munet_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cpp>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(munet_core PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME core)
if(MUNET_VULKAN)
  # CPU execution and owner startup must work without a Vulkan loader installed.
  find_path(Vulkan_INCLUDE_DIR vulkan/vulkan.h HINTS "$ENV{VULKAN_SDK}/include" REQUIRED)
  target_include_directories(munet_core PRIVATE ${Vulkan_INCLUDE_DIR})
  target_link_libraries(munet_core PRIVATE ${CMAKE_DL_LIBS})
  target_compile_definitions(munet_core PRIVATE MUNET_VULKAN=1)
endif()
if(MUNET_SWARM_NODE)
  if(NOT UNIX)
    message(FATAL_ERROR "The v1 swarm worker currently requires POSIX file locking and fsync")
  endif()
  find_package(CURL REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(munet-node cpp/swarm/node.cpp)
  target_include_directories(munet-node PRIVATE cpp/third_party)
  target_link_libraries(munet-node PRIVATE munet_core CURL::libcurl OpenSSL::Crypto Threads::Threads)
  target_compile_definitions(munet-node PRIVATE MUNET_VERSION="${MUNET_VERSION}")
  if(SKBUILD)
    install(TARGETS munet-node RUNTIME DESTINATION munet/bin)
  else()
    install(TARGETS munet-node RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  endif()
endif()
if(MUNET_INSTALL_SDK)
  include(CMakePackageConfigHelpers)
  install(TARGETS munet_core EXPORT MuNetTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(FILES cpp/core.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/munet)
  install(EXPORT MuNetTargets NAMESPACE MuNet:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MuNet)
  configure_package_config_file(cmake/MuNetConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MuNetConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MuNet)
  write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MuNetConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MuNetConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MuNetConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MuNet)
endif()
if(MUNET_PYTHON)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native cpp/bindings.cpp)
  target_link_libraries(_native PRIVATE munet_core)
  install(TARGETS _native LIBRARY DESTINATION munet RUNTIME DESTINATION munet)
endif()
